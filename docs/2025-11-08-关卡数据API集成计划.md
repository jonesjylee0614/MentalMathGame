### 步骤1：创建关卡数据Hook

**文件**: `frontend/src/hooks/useLevels.ts`

```typescript
import { useState, useEffect } from 'react';
import { levelsApi } from '../api/levels';
import { Level } from '../lib/types';

interface UseLevelsResult {
  levels: Level[];
  loading: boolean;
  error: string | null;
  refresh: () => Promise<void>;
}

export const useLevels = (category?: string): UseLevelsResult => {
  const [levels, setLevels] = useState<Level[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchLevels = async () => {
    try {
      setLoading(true);
      setError(null);
      const response = await levelsApi.getLevels({ category, status: 1 });
      
      if (response.success && response.data) {
        // 转换后端数据格式到前端格式
        const transformedLevels = response.data.map((level: any) => ({
          id: level.id,
          category: level.category,
          name: level.name,
          desc: level.description || '',
          generator: level.generator_config || {},
          count: level.question_count || 20,
          timeSec: level.time_limit || 120,
          targetTime: level.target_time || 0,
          difficulty: level.difficulty || 1.0,
          rewardPoints: level.reward_points || 0,
          gameMode: level.game_mode,
          recommendedModes: level.recommended_modes || [],
          modeConfig: level.mode_config || {},
        }));
        
        setLevels(transformedLevels);
      } else {
        setError(response.message || '加载关卡失败');
      }
    } catch (err) {
      setError(err instanceof Error ? err.message : '网络错误');
      console.error('Failed to fetch levels:', err);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchLevels();
  }, [category]);

  return {
    levels,
    loading,
    error,
    refresh: fetchLevels,
  };
};
```

**功能说明**：
- 自动加载关卡数据
- 支持按分类筛选
- 提供加载状态和错误处理
- 支持手动刷新
- 自动转换后端数据格式

### 步骤2：创建关卡详情Hook

**文件**: `frontend/src/hooks/useLevel.ts`

```typescript
import { useState, useEffect } from 'react';
import { levelsApi } from '../api/levels';
import { Level } from '../lib/types';

interface UseLevelResult {
  level: Level | null;
  loading: boolean;
  error: string | null;
}

export const useLevel = (levelId: string | undefined): UseLevelResult => {
  const [level, setLevel] = useState<Level | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (!levelId) {
      setLevel(null);
      setLoading(false);
      return;
    }

    const fetchLevel = async () => {
      try {
        setLoading(true);
        setError(null);
        const response = await levelsApi.getLevelDetail(levelId);
        
        if (response.success && response.data) {
          // 转换数据格式
          const transformedLevel: Level = {
            id: response.data.id,
            category: response.data.category,
            name: response.data.name,
            desc: response.data.description || '',
            generator: response.data.generator_config || {},
            count: response.data.question_count || 20,
            timeSec: response.data.time_limit || 120,
            targetTime: response.data.target_time || 0,
            difficulty: response.data.difficulty || 1.0,
            rewardPoints: response.data.reward_points || 0,
            gameMode: response.data.game_mode,
            recommendedModes: response.data.recommended_modes || [],
            modeConfig: response.data.mode_config || {},
          };
          
          setLevel(transformedLevel);
        } else {
          setError(response.message || '加载关卡失败');
        }
      } catch (err) {
        setError(err instanceof Error ? err.message : '网络错误');
        console.error('Failed to fetch level:', err);
      } finally {
        setLoading(false);
      }
    };

    fetchLevel();
  }, [levelId]);

  return { level, loading, error };
};
```

**功能说明**：
- 根据ID加载单个关卡详情
- 用于游戏页面
- 提供加载状态

```typescript
const categories = useMemo(() => {

import { useLevels } from '../hooks/useLevels';

const { levels, loading, error } = useLevels();

const categories = useMemo(() => {
  // 将levels按category分组
  const map = new Map<string, Level[]>();
  levels.forEach((level) => {
    if (!map.has(level.category)) map.set(level.category, []);
    map.get(level.category)!.push(level);
  });
  
  const allCategories = Array.from(map.entries()).map(([name, items]) => ({ 
    name, 
    items 
  }));
  
  // 应用搜索过滤
  const keyword = search.trim();
  if (!keyword) return allCategories;
  
  return allCategories
    .map((category) => ({
      ...category,
      items: category.items.filter((item) =>
        [item.name, item.desc, item.category, item.id].some((text) => 
          text.includes(keyword)
        )
      )
    }))
    .filter((category) => category.items.length > 0);
}, [levels, search]);

// 添加加载和错误状态显示
if (loading) {

    <div className="glass-card">
      <p>加载失败：{error}</p>
      <button className="btn" onClick={() => window.location.reload()}>
        重试
      </button>
```

### 步骤5：更新levels.ts工具函数

**文件**: `frontend/src/lib/levels.ts`

**修改内容**：
```typescript
// 保留LEVELS数组作为fallback（离线使用或开发环境）
export const LEVELS: Level[] = [
  // ... 保留现有数据作为备份
];

// 修改为从缓存或API获取
let cachedLevels: Level[] | null = null;

export const setLevels = (levels: Level[]) => {
  cachedLevels = levels;
};

export const findLevel = (levelId: string): Level | null => {
  const levels = cachedLevels || LEVELS;
  return levels.find((level) => level.id === levelId) ?? null;
};

export const listCategories = () => {
  const levels = cachedLevels || LEVELS;
  const map = new Map<string, Level[]>();
  levels.forEach((level) => {
    if (!map.has(level.category)) map.set(level.category, []);
    map.get(level.category)!.push(level);
  });
  return Array.from(map.entries()).map(([name, items]) => ({ name, items }));
};
```

**修改要点**：
- 保留LEVELS数组作为fallback
- 添加缓存机制
- 保持API兼容性

```typescript
import React, { createContext, useContext, useState, useEffect } from 'react';
import { levelsApi } from '../api/levels';
import { Level } from '../lib/types';
import { setLevels } from '../lib/levels';

interface LevelsContextValue {
  levels: Level[];
  loading: boolean;
  error: string | null;
  refresh: () => Promise<void>;
}

const LevelsContext = createContext<LevelsContextValue | undefined>(undefined);

export const LevelsProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [levels, setLevelsState] = useState<Level[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchLevels = async () => {
    try {
      setLoading(true);
      setError(null);
      const response = await levelsApi.getLevels({ status: 1 });
      
      if (response.success && response.data) {
        const transformedLevels = response.data.map((level: any) => ({
          // ... 转换逻辑
        }));
        
        setLevelsState(transformedLevels);
        setLevels(transformedLevels); // 更新全局缓存
      }
    } catch (err) {
      setError(err instanceof Error ? err.message : '网络错误');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchLevels();
  }, []);

  return (
    <LevelsContext.Provider value={{ levels, loading, error, refresh: fetchLevels }}>
      {children}
    </LevelsContext.Provider>
  );
};

export const useLevelsContext = () => {
  const context = useContext(LevelsContext);
  if (!context) {
    throw new Error('useLevelsContext must be used within LevelsProvider');
  }
  return context;
};
```

**功能说明**：
- 全局关卡数据管理
- 避免重复请求
- 统一加载状态
